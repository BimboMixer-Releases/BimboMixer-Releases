import re

with open('lib/screens/payments_screen.dart', 'r', encoding='utf-8') as f:
    content = f.read()

# Replace Expanded(child: Container(...))
content = re.sub(r'Expanded\(\s*child: Container\(', r'Container(', content, 1)

# Inside the Container, there's a Column. Let's find Expanded(child: LayoutBuilder(
content = re.sub(r'Expanded\(\s*child: LayoutBuilder\(', r'LayoutBuilder(', content, 1)

# Then there is Expanded(child: filtered.isEmpty
content = re.sub(r'Expanded\(\s*child: filtered\.isEmpty', r'filtered.isEmpty', content, 1)

# Now we need to add shrinkWrap and physics to ListView.separated
content = content.replace(
    'ListView.separated(',
    'ListView.separated( shrinkWrap: true, physics: const NeverScrollableScrollPhysics(),'
)

# And we need to wrap the outermost Column in SingleChildScrollView
content = content.replace(
    '''child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [''',
    '''child: SingleChildScrollView(
                physics: const BouncingScrollPhysics(),
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: ['''
)

# Because we added SingleChildScrollView, we added a bracket. But we removed 3 Expanded() widgets!
